 "use client";

import { Canvas, useFrame } from "@react-three/fiber";
import { useGLTF, OrbitControls } from "@react-three/drei";
import { Suspense, useRef } from "react";
import * as THREE from "three";

function HardwareModel({ path, hovered }: {
  path: string;
  hovered: boolean;
}) {
  const { scene } = useGLTF(path);
  const model = useRef<THREE.Group>(null);

  useFrame((state, delta) => {
    if (!model.current) return;

    model.current.rotation.y += delta * (hovered ? 0.7 : 0.12);

    const targetScale = hovered ? 1.25 : 1;

    model.current.scale.lerp(
      new THREE.Vector3(targetScale, targetScale, targetScale),
      1 - Math.exp(-8 * delta)
    );

    model.current.position.y = THREE.MathUtils.damp(
      model.current.position.y,
      hovered ? 0.2 : 0,
      8,
      delta
    );
  });

  return (
    <group ref={model}>
      <primitive object={scene} />
    </group>
  );
}

export default function InteractiveLabModel({
  modelPath,
  hovered
}: {
  modelPath: string;
  hovered: boolean;
}) {
  return (
    <div className="h-full w-full bg-[#FFF8E8]">
      <Canvas camera={{ position: [3, 2, 4], fov: 40 }}>
        <ambientLight intensity={1.5} />
        <directionalLight position={[4, 5, 5]} intensity={2} />
        <Suspense fallback={null}>
          <HardwareModel path={modelPath} hovered={hovered} />
          <OrbitControls enableZoom={false} />
        </Suspense>
      </Canvas>
    </div>
  );
